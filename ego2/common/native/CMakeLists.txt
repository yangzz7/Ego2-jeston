cmake_minimum_required(VERSION 3.18)
project(ego2_cyper_cuda LANGUAGES CXX CUDA)
set(CMAKE_CUDA_STANDARD 14)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 87)
endif()
add_library(ego2_cyper_cuda SHARED cyper_cuda.cu)
set_target_properties(ego2_cyper_cuda PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
target_compile_options(ego2_cyper_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
