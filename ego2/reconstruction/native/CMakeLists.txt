cmake_minimum_required(VERSION 3.16)
project(ego2_reconstruction LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
# Build every Eigen-using library with the same architecture/alignment flags.
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG -march=native")
set(CMAKE_BUILD_RPATH_USE_ORIGIN TRUE)
# Keep runtime lookup relocatable even though bin/lib sit outside the CMake
# cache directory. /usr/local/lib contains the system Pangolin installation.
set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib;/usr/local/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${EGO2_OUTPUT}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${EGO2_OUTPUT}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${EGO2_OUTPUT}/cmake/lib")
if(EXISTS "/usr/lib/x86_64-linux-gnu/cmake/opencv4/OpenCVConfig.cmake")
  set(OpenCV_DIR "/usr/lib/x86_64-linux-gnu/cmake/opencv4" CACHE PATH "OpenCV config" FORCE)
endif()
find_package(OpenCV 4.2 REQUIRED)
find_package(Eigen3 3.1 REQUIRED)
find_package(Pangolin REQUIRED)
find_package(Threads REQUIRED)
find_library(ZMQ_LIBRARY zmq REQUIRED)
add_subdirectory(${ORB_SOURCE}/Thirdparty/DBoW2 ${CMAKE_BINARY_DIR}/DBoW2)
set(g2o_LIBRARY_OUTPUT_DIRECTORY "${EGO2_OUTPUT}/lib" CACHE PATH "g2o library output" FORCE)
add_subdirectory(${ORB_SOURCE}/Thirdparty/g2o ${CMAKE_BINARY_DIR}/g2o)
foreach(lib DBoW2 g2o)
  set_target_properties(${lib} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${EGO2_OUTPUT}/lib")
endforeach()
include_directories(${ORB_SOURCE} ${ORB_SOURCE}/include ${ORB_SOURCE}/include/CameraModels
  ${ORB_SOURCE}/Thirdparty/Sophus ${EIGEN3_INCLUDE_DIR} ${Pangolin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
set(ORB_FILES System Tracking LocalMapping LoopClosing ORBextractor ORBmatcher FrameDrawer Converter
  MapPoint KeyFrame Atlas Map MapDrawer Optimizer Frame KeyFrameDatabase Sim3Solver Viewer ImuTypes G2oTypes
  GeometricTools TwoViewReconstruction Config Settings)
foreach(name IN LISTS ORB_FILES)
  list(APPEND ORB_CPP ${ORB_SOURCE}/src/${name}.cc)
endforeach()
list(APPEND ORB_CPP ${ORB_SOURCE}/src/CameraModels/Pinhole.cpp
  ${ORB_SOURCE}/src/CameraModels/KannalaBrandt8.cpp ${ORB_SOURCE}/src/OptimizableTypes.cpp
  ${ORB_SOURCE}/src/MLPnPsolver.cpp)
add_library(ego2_orb STATIC ${ORB_CPP} snapshot.cc)
target_link_libraries(ego2_orb PUBLIC ${OpenCV_LIBS} ${Pangolin_LIBRARIES}
  DBoW2 g2o boost_serialization crypto Threads::Threads)
add_executable(ego2_reconstruction_native capture.cc ${ORB_SOURCE}/Thirdparty/usb/uvc/linux/uvc-v4l2.cc)
target_link_libraries(ego2_reconstruction_native PRIVATE ego2_orb ${ZMQ_LIBRARY} ${CMAKE_DL_LIBS})
add_executable(ego2_multi_vio_native capture.cc ${ORB_SOURCE}/Thirdparty/usb/uvc/linux/uvc-v4l2.cc)
target_compile_definitions(ego2_multi_vio_native PRIVATE EGO2_MULTI_VIO=1)
target_link_libraries(ego2_multi_vio_native PRIVATE ego2_orb ${ZMQ_LIBRARY} ${CMAKE_DL_LIBS})
enable_testing()
add_executable(test_imu_interval test_imu_interval.cc)
target_link_libraries(test_imu_interval PRIVATE ${OpenCV_LIBS})
add_test(NAME imu_interval COMMAND test_imu_interval)
add_executable(test_multi_vio_transport ../../multi_vio/native/test_transport.cc)
target_link_libraries(test_multi_vio_transport PRIVATE ${OpenCV_LIBS})
add_test(NAME multi_vio_transport COMMAND test_multi_vio_transport)
add_executable(test_vio_preview_bundle ../../multi_vio/native/test_preview_bundle.cc)
target_link_libraries(test_vio_preview_bundle PRIVATE ${OpenCV_LIBS} Threads::Threads)
add_executable(test_multi_vio_archive ../../multi_vio/native/test_archive.cc)
target_link_libraries(test_multi_vio_archive PRIVATE ${OpenCV_LIBS} Threads::Threads)
add_test(NAME multi_vio_archive COMMAND test_multi_vio_archive)
add_executable(test_display_raw test_display_raw.cc)
target_link_libraries(test_display_raw PRIVATE ${OpenCV_LIBS})
add_test(NAME display_raw COMMAND test_display_raw)
add_executable(test_capture_cuda test_capture_cuda.cc)
target_link_libraries(test_capture_cuda PRIVATE ${OpenCV_LIBS} ${CMAKE_DL_LIBS} Threads::Threads)
# Explicit opt-in: this test requires a CUDA-capable host and the capture library.
